using System;
using System.IO;
using System.Linq;
using System.Text;
using System.Windows.Markup;
using System.Xml;

// 用 XamlReader 真正解析 XAML，把任何结构错误连行号一起报出来。
// 比字符串扫描可靠 —— Python 侧扫不出「标签起始丢失」这类错误。
class XC
{
    [STAThread]
    static int Main(string[] a)
    {
        string path = a.Length > 0 ? a[0] : "../_xamlcheck.xaml";
        string xaml = File.ReadAllText(path, Encoding.UTF8);
        var sb = new StringBuilder();
        int code = 0;
        try
        {
            using (var sr = new StringReader(xaml))
            using (var xr = System.Xml.XmlReader.Create(sr))
            {
                var obj = XamlReader.Load(xr);
                sb.AppendLine("[OK] XAML 解析成功（对象已构造）");
                sb.AppendLine("  类型 = " + (obj == null ? "null" : obj.GetType().FullName));
            }
        }
        catch (XmlException ex)
        {
            code = 1;
            sb.AppendLine("[FAIL] XML 结构错误");
            sb.AppendLine("  行号 " + ex.LineNumber + "  位置 " + ex.LinePosition);
            sb.AppendLine("  " + ex.Message);
            // 打印出错行附近
            var lines = xaml.Split('\n');
            for (int i = Math.Max(0, ex.LineNumber - 4); i < Math.Min(lines.Length, ex.LineNumber + 3); i++)
                sb.AppendLine("   " + (i + 1 == ex.LineNumber ? " >> " : "    ") + (i + 1) + " | " + lines[i].Trim());
        }
        catch (Exception ex)
        {
            code = 2;
            sb.AppendLine("[FAIL] " + ex.GetType().Name);
            sb.AppendLine("  " + ex.Message);
            if (ex.InnerException != null) sb.AppendLine("  inner: " + ex.InnerException.Message);
        }
        File.WriteAllText("_xc-out.txt", sb.ToString(), new UTF8Encoding(false));
        Console.OutputEncoding = Encoding.UTF8;
        Console.WriteLine(sb.ToString());
        return code;
    }
}
