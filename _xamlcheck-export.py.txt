"""把 C# 里的 XAML 字符串抽出来，用 .NET 的 XamlReader 真正解析一遍。
比 Python 字符串扫描可靠得多 —— 能直接报出「哪一行结构坏了」。

用法：dotnet run 或 python 调 XamlCheck.exe
"""
import subprocess, sys, os, re, io

CS = 'FurinaSettings.cs'

def extract_xaml():
    s = io.open(CS, encoding='utf-8').read()
    # XamlText = @"..."  是逐字字符串（"" 表示一个 "）
    m = re.search(r'const string XamlText\s*=\s*@"', s)
    if not m:
        raise SystemExit('找不到 XamlText')
    i = m.end()
    out = []
    while i < len(s):
        c = s[i]
        if c == '"':
            if i + 1 < len(s) and s[i+1] == '"':
                out.append('"'); i += 2; continue
            break
        out.append(c); i += 1
    return ''.join(out)

if __name__ == '__main__':
    xaml = extract_xaml()
    out = '_xamlcheck.xaml'
    io.open(out, 'w', encoding='utf-8').write(xaml)
    print('XAML 已导出: %s (%d 行, %d 字符)' % (out, xaml.count('\n')+1, len(xaml)))
